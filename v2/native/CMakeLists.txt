project(MaleModWitcherV2 LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(MALEMOD_MINHOOK_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../build/research/minhook" CACHE PATH "Pinned engine hook checkout")
set(MALEMOD_BASE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../../MaleMod" CACHE PATH "Base interface checkout")

if(NOT hook STREQUAL "8af6b4acae5a9388fd742b56fa79ece89d96f823")

endif()

target_compile_definitions(malemod_witcher PRIVATE MALEMOD_OWNED_PANEL_PROBE=1)
target_compile_features(malemod_witcher PRIVATE cxx_std_17)
target_compile_options(malemod_witcher PRIVATE /W4 /fp:precise /utf-8)
target_link_libraries(malemod_witcher PRIVATE minhook bcrypt dxguid d3d12 d3dcompiler user32 gdi32 winmm)
