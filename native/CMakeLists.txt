cmake_minimum_required(VERSION 3.16)
project(MaleModWitcherNative LANGUAGES C CXX)
set(MALEMOD_BASE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../MaleMod" CACHE PATH "Pinned Base checkout")
set(MALEMOD_SURFACE_RECIPE "process-parallel" CACHE STRING "Verified Base numerical recipe")
if(MALEMOD_SURFACE_RECIPE STREQUAL "process-parallel")
 set(surface_build "surface-process-verified-cmake")
 set(surface_checks "processSourceChecks")
elseif(MALEMOD_SURFACE_RECIPE STREQUAL "tls")
 set(surface_build "surface-cmake")
 set(surface_checks "currentSourceChecks")
else()
 message(FATAL_ERROR "Unknown Base numerical recipe")
endif()
set(MALEMOD_SURFACE_LIBRARY "${MALEMOD_BASE_PATH}/build/${surface_build}/Release/malemod_surface.lib" CACHE FILEPATH "Verified Win32 numerical library")
option(MALEMOD_ALLOW_DIRTY_BASE_DIAGNOSTIC "Explicit developer experiment against an uncommitted Base" OFF)
option(MALEMOD_ALLOW_UNVERIFIED_SURFACE_LIBRARY "Explicit diagnostic opt-in; never a deployment library" OFF)
find_package(Git REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../dependencies/base.lock.json" base_lock)
string(JSON expected_base GET "${base_lock}" commit)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MALEMOD_BASE_PATH}" rev-parse HEAD OUTPUT_VARIABLE actual_base OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE base_result)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MALEMOD_BASE_PATH}" status --porcelain --untracked-files=no OUTPUT_VARIABLE base_dirty OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE dirty_result)
if(NOT base_result EQUAL 0 OR NOT dirty_result EQUAL 0)
 message(FATAL_ERROR "Cannot verify the Base checkout")
endif()
if((NOT actual_base STREQUAL expected_base OR NOT base_dirty STREQUAL "") AND NOT MALEMOD_ALLOW_DIRTY_BASE_DIAGNOSTIC)
 message(FATAL_ERROR "Base differs from the committed adapter pin. Commit/pin first; developer experiments require explicit diagnostic opt-in.")
endif()
include(CTest)
if(NOT WIN32)
 message(FATAL_ERROR "This target is the Windows engine/process adapter")
endif()
add_executable(surface_wire_test "${MALEMOD_BASE_PATH}/tests/surface_wire_test.cpp")
target_include_directories(surface_wire_test PRIVATE "${MALEMOD_BASE_PATH}/include")
target_compile_features(surface_wire_test PRIVATE cxx_std_17)
add_test(NAME surface_wire COMMAND surface_wire_test)
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
 file(READ "${MALEMOD_BASE_PATH}/provenance/source-surface.json" surface_provenance)
 string(JSON expected_library GET "${surface_provenance}" ${surface_checks} librarySHA256)
 file(SHA256 "${MALEMOD_SURFACE_LIBRARY}" actual_library)
 if(NOT actual_library STREQUAL expected_library AND NOT MALEMOD_ALLOW_UNVERIFIED_SURFACE_LIBRARY)
  message(FATAL_ERROR "Win32 surface library differs from verified Base provenance; rebuild/verify before adoption")
 endif()
 add_executable(surface_worker surface_worker.cpp)
 target_link_libraries(surface_worker PRIVATE "${MALEMOD_SURFACE_LIBRARY}")
 target_include_directories(surface_worker PRIVATE "${MALEMOD_BASE_PATH}/include")
 target_compile_features(surface_worker PRIVATE cxx_std_17)
 target_compile_options(surface_worker PRIVATE /fp:strict /utf-8)
 set_property(TARGET surface_worker PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
else()
 set(MALEMOD_GRAFT_LIBRARY "${MALEMOD_BASE_PATH}/build/graft-runtime-cmake/Release/malemod_graft.lib" CACHE FILEPATH "Verified x64 Base target collar")
 file(READ "${MALEMOD_BASE_PATH}/provenance/graft-runtime.json" graft_provenance)
 string(JSON expected_graft GET "${graft_provenance}" librarySHA256)
 file(SHA256 "${MALEMOD_GRAFT_LIBRARY}" actual_graft)
 if(NOT actual_graft STREQUAL expected_graft)
  message(FATAL_ERROR "Target collar library differs from verified Base provenance")
 endif()
 add_library(malemod_target_surface STATIC target_surface.cpp)
 target_include_directories(malemod_target_surface PUBLIC "${MALEMOD_BASE_PATH}/include")
 target_link_libraries(malemod_target_surface PUBLIC "${MALEMOD_GRAFT_LIBRARY}")
 target_compile_features(malemod_target_surface PUBLIC cxx_std_17)
 target_compile_options(malemod_target_surface PRIVATE /fp:strict /W4 /utf-8)
 set_property(TARGET malemod_target_surface PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
 add_executable(target_surface_test target_surface_test.cpp)
 target_link_libraries(target_surface_test PRIVATE malemod_target_surface)
 set_property(TARGET target_surface_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
 add_executable(surface_pipeline_test surface_pipeline_test.cpp)
 target_link_libraries(surface_pipeline_test PRIVATE malemod_target_surface)
 target_compile_options(surface_pipeline_test PRIVATE /fp:strict /W4 /utf-8)
 set_property(TARGET surface_pipeline_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
 add_executable(surface_service_test surface_service_test.cpp)
 target_link_libraries(surface_service_test PRIVATE malemod_target_surface)
 target_compile_options(surface_service_test PRIVATE /fp:strict /W4 /utf-8)
 set_property(TARGET surface_service_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
 set(MALEMOD_MINHOOK_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../build/research/minhook" CACHE PATH "Pinned native hook dependency")
 if(EXISTS "${MALEMOD_MINHOOK_PATH}/CMakeLists.txt")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../dependencies/native.lock.json" native_lock)
  string(JSON expected_hook GET "${native_lock}" dependencies 0 commit)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MALEMOD_MINHOOK_PATH}" rev-parse HEAD OUTPUT_VARIABLE actual_hook OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE hook_result)
  if(NOT hook_result EQUAL 0 OR NOT actual_hook STREQUAL expected_hook)
   message(FATAL_ERROR "MinHook differs from the native dependency pin")
  endif()
  add_subdirectory("${MALEMOD_MINHOOK_PATH}" "minhook")
  set_property(TARGET minhook PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
  add_library(malemod_witcher SHARED engine_bridge.cpp graphics_probe.cpp)
  target_include_directories(malemod_witcher PRIVATE "${MALEMOD_BASE_PATH}/include")
  target_link_libraries(malemod_witcher PRIVATE minhook bcrypt dxguid)
  target_compile_features(malemod_witcher PRIVATE cxx_std_17)
  target_compile_options(malemod_witcher PRIVATE /W4 /utf-8)
  set_property(TARGET malemod_witcher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
  add_executable(graphics_probe_test graphics_probe_test.cpp graphics_probe.cpp)
  target_link_libraries(graphics_probe_test PRIVATE minhook d3d12 dxgi dxguid)
  target_compile_features(graphics_probe_test PRIVATE cxx_std_17)
  target_compile_options(graphics_probe_test PRIVATE /W4 /utf-8)
  set_property(TARGET graphics_probe_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
  add_test(NAME graphics_probe_sdk COMMAND graphics_probe_test)
  add_executable(launch_native launch_native.cpp)
  target_link_libraries(launch_native PRIVATE psapi)
  target_compile_features(launch_native PRIVATE cxx_std_17)
  target_compile_options(launch_native PRIVATE /W4 /utf-8)
  set_property(TARGET launch_native PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
 endif()
 add_executable(surface_client_test surface_client_test.cpp)
 target_include_directories(surface_client_test PRIVATE "${MALEMOD_BASE_PATH}/include")
 target_compile_features(surface_client_test PRIVATE cxx_std_17)
 target_compile_options(surface_client_test PRIVATE /fp:strict /utf-8)
 set_property(TARGET surface_client_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
