cmake_minimum_required(VERSION 3.20)
project(MaleModWitcherSandboxLauncher LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
 message(FATAL_ERROR "The private Windows studio requires MSVC and Windows SDK")
endif()
add_executable(launch_sealed ../../native/launch_sealed.cpp)
target_compile_features(launch_sealed PRIVATE cxx_std_17)
target_compile_options(launch_sealed PRIVATE /W4 /utf-8)
target_link_libraries(launch_sealed PRIVATE ole32 uuid)
set_property(TARGET launch_sealed PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
set_target_properties(launch_sealed PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}")
