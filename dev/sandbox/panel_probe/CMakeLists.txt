cmake_minimum_required(VERSION 3.16)
project(MaleModOwnedPanelProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "The scoped panel fixture requires the x64 Windows SDK")
endif()
set(MALEMOD_BASE_PATH "" CACHE PATH "Exact clean shared theme/header checkout for this SDK fixture")
set(MALEMOD_BASE_REVISION "" CACHE STRING "Required source commit; not a replacement installed-game pin")
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MALEMOD_BASE_PATH}" rev-parse HEAD OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE result)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MALEMOD_BASE_PATH}" status --porcelain --untracked-files=no -- include OUTPUT_VARIABLE dirty OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE status_result)
if(NOT result EQUAL 0 OR NOT status_result EQUAL 0 OR NOT actual STREQUAL MALEMOD_BASE_REVISION OR NOT dirty STREQUAL "" OR MALEMOD_BASE_REVISION STREQUAL "")
 message(FATAL_ERROR "Provide the exact clean Base header revision explicitly")
endif()
set(native "${CMAKE_CURRENT_SOURCE_DIR}/../../../native")
add_executable(overlay_panel_probe_test fixture.cpp "${native}/overlay_panel.cpp")
target_include_directories(overlay_panel_probe_test PRIVATE "${MALEMOD_BASE_PATH}/include" "${native}")
target_compile_definitions(overlay_panel_probe_test PRIVATE MALEMOD_OWNED_PANEL_PROBE=1)
target_link_libraries(overlay_panel_probe_test PRIVATE user32 gdi32)
add_executable(overlay_panel_production_test "${native}/overlay_panel_test.cpp" "${native}/overlay_panel.cpp")
target_include_directories(overlay_panel_production_test PRIVATE "${MALEMOD_BASE_PATH}/include")
target_link_libraries(overlay_panel_production_test PRIVATE user32 gdi32)
add_executable(launch_sealed "${native}/launch_sealed.cpp")
target_link_libraries(launch_sealed PRIVATE user32 ole32)
foreach(target overlay_panel_probe_test overlay_panel_production_test launch_sealed)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_options(${target} PRIVATE /W4 /utf-8)
 set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endforeach()
